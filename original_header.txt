function HeaderNavItem({
  id,
  label,
  isActive,
  onClick,
}: {
  id: string;
  label: string;
  isActive: boolean;
  isDarkBackground?: boolean;
  onClick: () => void;
}) {
  const activeLineRef = useRef<HTMLDivElement>(null);
  const hoverLineRef = useRef<HTMLDivElement>(null);
  const textRef = useRef<HTMLSpanElement>(null);
  const isMountedRef = useRef(false);

  useEffect(() => {
    return () => {
      if (activeLineRef.current) gsap.killTweensOf(activeLineRef.current);
      if (hoverLineRef.current) gsap.killTweensOf(hoverLineRef.current);
      if (textRef.current) gsap.killTweensOf(textRef.current);
    };
  }, []);

  useEffect(() => {
    if (!activeLineRef.current) return;

    if (!isMountedRef.current) {
      isMountedRef.current = true;
      gsap.set(activeLineRef.current, {
        scaleX: isActive ? 1 : 0,
        transformOrigin: "left",
      });
      if (hoverLineRef.current) {
        gsap.set(hoverLineRef.current, { scaleX: 0, transformOrigin: "left" });
      }
      return;
    }

    gsap.killTweensOf(activeLineRef.current);
    if (isActive) {
      if (hoverLineRef.current) {
        gsap.killTweensOf(hoverLineRef.current);
        gsap.set(hoverLineRef.current, { scaleX: 0 });
      }
      gsap.fromTo(
        activeLineRef.current,
        { scaleX: 0, transformOrigin: "left" },
        { scaleX: 1, duration: 0.42, ease: "power3.out" }
      );
    } else {
      gsap.to(activeLineRef.current, {
        scaleX: 0,
        transformOrigin: "right",
        duration: 0.32,
        ease: "power2.inOut",
      });
    }
  }, [isActive]);

  const handleMouseEnter = () => {
    if (isActive) return;
    if (hoverLineRef.current) {
      gsap.killTweensOf(hoverLineRef.current);
      gsap.fromTo(
        hoverLineRef.current,
        { scaleX: 0, transformOrigin: "left" },
        { scaleX: 1, duration: 0.32, ease: "power2.out" }
      );
    }
    if (textRef.current) {
      gsap.killTweensOf(textRef.current);
      gsap.to(textRef.current, {
        y: -1.5,
        duration: 0.22,
        ease: "power2.out",
      });
    }
  };

  const handleMouseLeave = () => {
    if (isActive) return;
    if (hoverLineRef.current) {
      gsap.killTweensOf(hoverLineRef.current);
      gsap.to(hoverLineRef.current, {
        scaleX: 0,
        transformOrigin: "right",
        duration: 0.28,
        ease: "power2.inOut",
      });
    }
    if (textRef.current) {
      gsap.killTweensOf(textRef.current);
      gsap.to(textRef.current, {
        y: 0,
        duration: 0.22,
        ease: "power2.out",
      });
    }
  };

  return (
    <button
      id={id}
      type="button"
      onClick={onClick}
      onMouseEnter={handleMouseEnter}
      onMouseLeave={handleMouseLeave}
      className="group relative flex flex-col items-center py-1 px-1 cursor-pointer select-none"
    >
      <span
        ref={textRef}
        className={`font-mono text-xs sm:text-sm tracking-widest uppercase text-white will-change-transform transition-opacity duration-200 ${
          isActive
            ? "font-bold opacity-100"
            : "font-medium opacity-65 group-hover:opacity-100"
        }`}
      >
        {label}
      </span>
      <div className="relative w-full h-[2.5px] sm:h-[3px] mt-1 overflow-hidden rounded-[1px]">
        <div className="absolute inset-0 bg-white/20" />
        <div
          ref={hoverLineRef}
          className="absolute inset-0 origin-left will-change-transform pointer-events-none bg-white/80"
        />
        <div
          ref={activeLineRef}
          className="absolute inset-0 origin-left will-change-transform pointer-events-none bg-white"
        />
      </div>
    </button>
  );
}

const GLITCH_GLYPHS = "!<>-_\\/[]{}—=+*^?#_01";

function HeaderLogo() {
  const [isAlt, setIsAlt] = useState(false);
  const [text, setText] = useState<{ prefix: string; suffix: string }>({
    prefix: "markappscript",
    suffix: ".dev",
  });
  const [isGlitching, setIsGlitching] = useState(false);
  const animFrameRef = useRef<number | null>(null);
  const btnRef = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    return () => {
      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);
    };
  }, []);

  const handleClick = (e: React.MouseEvent) => {
    e.stopPropagation();
    if (isGlitching) return;
    setIsGlitching(true);

    const nextIsAlt = !isAlt;
    setIsAlt(nextIsAlt);

    const targetPrefix = nextIsAlt ? "Code by Mark" : "markappscript";
    const targetSuffix = nextIsAlt ? "©" : ".dev";
    const fullTarget = targetPrefix + targetSuffix;

    if (btnRef.current) {
      gsap.killTweensOf(btnRef.current);
      gsap.timeline()
        .to(btnRef.current, { x: -2, y: 1, duration: 0.04, ease: "none" })
        .to(btnRef.current, { x: 2.5, y: -1, duration: 0.04, ease: "none" })
        .to(btnRef.current, { x: -1.5, y: 0.5, duration: 0.04, ease: "none" })
        .to(btnRef.current, { x: 1, y: -0.5, duration: 0.04, ease: "none" })
        .to(btnRef.current, { x: 0, y: 0, duration: 0.04, ease: "none" });
    }

    let frame = 0;
    const totalFrames = 12;

    const animate = () => {
      frame++;
      const progress = frame / totalFrames;

      if (frame >= totalFrames) {
        setText({ prefix: targetPrefix, suffix: targetSuffix });
        setIsGlitching(false);
        return;
      }

      let scrambled = "";
      for (let i = 0; i < fullTarget.length; i++) {
        if (progress > i / fullTarget.length + 0.18) {
          scrambled += fullTarget[i];
        } else {
          scrambled += GLITCH_GLYPHS[Math.floor(Math.random() * GLITCH_GLYPHS.length)];
        }
      }

      const splitIndex = targetPrefix.length;
      setText({
        prefix: scrambled.slice(0, splitIndex),
        suffix: scrambled.slice(splitIndex),
      });

      animFrameRef.current = requestAnimationFrame(animate);
    };

    animFrameRef.current = requestAnimationFrame(animate);
  };

  return (
    <Magnetic strength={0.2} className="justify-center">
      <button
        ref={btnRef}
        type="button"
        onClick={handleClick}
        className="flex items-baseline justify-center text-center select-none cursor-pointer will-change-transform min-w-[125px] sm:min-w-[160px]"
        aria-label="Toggle logo title"
      >
        <span
          className={`font-bold tracking-tight text-xs sm:text-base text-white mix-blend-difference ${
            isGlitching ? "drop-shadow-[1.5px_0_0_rgba(147,51,234,0.8)]" : ""
          }`}
        >
          {text.prefix}
        </span>
        <span
          className={`font-bold tracking-tight text-xs sm:text-base ${
            isAlt ? "text-white mix-blend-difference" : "text-purple-600"
          } ${isGlitching ? "drop-shadow-[-1.5px_0_0_rgba(6,182,212,0.8)]" : ""}`}
        >
          {text.suffix}
        </span>
      </button>
    </Magnetic>
  );
}
